## Complete Example: Updated AssessmentScoringTab

```typescript
import { useCallback, useEffect, useMemo, useState } from "react";
import { useScoringAgent } from "../hooks/useScoringAgent.js";
import { useSavedChangesToast } from "../context/SavedChangesToastContext.js";
import { Alert, AlertTitle } from "@mui/material";

export default function AssessmentScoringTab({
  assessmentName = "",
  includedAssetIds,
  excludedScopeCyberRiskIds,
  excludedScopeScenarioIds,
  scenarioNotApplicableIds,
  // ... other props
}: AssessmentScoringTabProps) {
  const navigate = useNavigate();
  const { notifySavedChanges } = useSavedChangesToast();
  
  // Add scoring agent hook
  const scoringAgent = useScoringAgent();

  // Build scoring rows as before
  const scoringRows = useMemo(
    () =>
      buildScoringRowsForScope(
        includedAssetIds,
        excludedScopeCyberRiskIds,
        excludedScopeScenarioIds,
        scenarioNotApplicableIds,
      ),
    [includedAssetIds, excludedScopeCyberRiskIds, excludedScopeScenarioIds, scenarioNotApplicableIds],
  );

  // Handle AI scoring start
  const handleStartScoring = useCallback(() => {
    const scenarioIds = scoringRows
      .filter((row) => row.kind === "scenario" && !scenarioNotApplicableIds.has(row.id))
      .map((row) => row.id);

    if (scenarioIds.length === 0) {
      alert("No scenarios to score");
      return;
    }

    scoringAgent.startScoring(
      scenarioIds,
      "ASM-001", // TODO: Get from props/context
      assessmentName
    );
  }, [scoringRows, scenarioNotApplicableIds, scoringAgent, assessmentName]);

  // Map agent phase to UI phase
  const aiScoringPhase: AiScoringPhase = useMemo(() => {
    switch (scoringAgent.phase) {
      case "processing":
        return "processing";
      case "complete":
        return "complete";
      case "error":
        return "idle"; // Reset to allow retry
      default:
        return "idle";
    }
  }, [scoringAgent.phase]);

  // Show success notification
  useEffect(() => {
    if (scoringAgent.phase === "complete" && scoringAgent.results) {
      const { summary } = scoringAgent.results;
      notifySavedChanges();

      if (summary.flaggedForReview > 0) {
        console.warn(
          `${summary.flaggedForReview} of ${summary.total} scenarios flagged for review`
        );
      }
    }
  }, [scoringAgent.phase, scoringAgent.results, notifySavedChanges]);

  return (
    <Stack sx={{ width: "100%", gap: 3, pt: 3, pb: 4 }}>
      {/* Error alert */}
      {scoringAgent.error && (
        <Alert severity="error" onClose={scoringAgent.reset}>
          <AlertTitle>Scoring Error</AlertTitle>
          {scoringAgent.error}
        </Alert>
      )}

      {/* Scoring info card */}
      <ScoringInfoCard
        omitHeader={aiScoringPhase === "idle"}
        title={aiScoringPhase === "complete" ? "AI scoring completed" : "AI scoring"}
        description={
          aiScoringPhase === "complete"
            ? `Successfully scored ${scoringAgent.results?.summary.succeeded || 0} scenarios`
            : "Our AI agent will automatically score all the scenarios."
        }
        onAction={aiScoringPhase === "idle" ? handleStartScoring : undefined}
        actionLoading={aiScoringPhase === "processing"}
        aggregationMethodRadio={{
          name: "cra-scoring-tab-aggregation",
          value: aggregationMethod,
          onValueChange: onAggregationMethodChange,
          disabled: aiScoringPhase === "processing",
        }}
      />

      {/* Scoring table */}
      {/* ... rest of component */}
    </Stack>
  );
}
```

## Troubleshooting

### Issue: Scores Don't Persist

**Fix**: Ensure `patchScenario` is called in hook (see Step 4)

### Issue: "Start AI scoring" Does Nothing

**Fix**: Check console for errors, verify `handleStartScoring` is wired up

### Issue: Skeleton Loaders Don't Show

**Fix**: Verify `aiScoringPhase === "processing"` is true during scoring

### Issue: Rationale Not Showing

**Fix**: Check `combinedRationaleSummary` field is populated and passed to WYSIWYG editor
